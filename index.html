async function playStation(id) { const allKnown = [...stations, ...apiSearchResults]; const s = allKnown.find(item => item.id === id); if (!s || !s.url) { console.error("❌ Emisora sin URL:", s); return; } console.log("🎵 Intentando reproducir:", s.name); console.log("🔗 Stream:", s.url); currentStationId = s.id; document.getElementById('currentTitle').innerText = s.name; document.getElementById('currentSub').innerText = `${s.city}, ${s.country}`; // Actualizar interfaz renderStations( document.getElementById('searchInput').value.trim() ? [...stations, ...apiSearchResults].filter(st => st.name.toLowerCase().includes( document.getElementById('searchInput').value.toLowerCase() ) ) : stations ); if (document.querySelector('.page:not(.hidden)').id === 'page-favglobal') { renderFavorites(); } // --------------------------------------------- // DETENER REPRODUCCIÓN ANTERIOR // --------------------------------------------- try { audio.pause(); } catch (e) { console.warn("No se pudo pausar audio anterior:", e); } if (hlsInstance) { try { hlsInstance.destroy(); } catch (e) { console.warn("Error destruyendo HLS:", e); } hlsInstance = null; } audio.removeAttribute('src'); try { audio.load(); } catch (e) { console.warn("Error haciendo load:", e); } // --------------------------------------------- // DETECTAR TIPO DE STREAM // --------------------------------------------- const url = s.url.trim(); const isHLS = url.toLowerCase().includes('.m3u8') || url.toLowerCase().includes('m3u8'); console.log( isHLS ? "📡 Stream detectado: HLS" : "🎧 Stream detectado: AUDIO DIRECTO" ); // --------------------------------------------- // HLS // --------------------------------------------- if (isHLS) { // Navegadores con HLS.js if (typeof Hls !== 'undefined' && Hls.isSupported()) { console.log("📡 Inicializando HLS.js..."); hlsInstance = new Hls({ enableWorker: true, lowLatencyMode: false, backBufferLength: 30, maxBufferLength: 30, manifestLoadingTimeOut: 10000, manifestLoadingMaxRetry: 3, manifestLoadingRetryDelay: 1000 }); hlsInstance.attachMedia(audio); hlsInstance.on( Hls.Events.MEDIA_ATTACHED, function () { console.log("✅ Audio conectado a HLS.js"); hlsInstance.loadSource(url); } ); hlsInstance.on( Hls.Events.MANIFEST_PARSED, async function () { console.log("✅ Manifest HLS cargado"); try { await audio.play(); console.log("▶️ REPRODUCIENDO:", s.name); document.getElementById('currentSub').innerText = `🔴 EN VIVO · ${s.city}, ${s.country}`; } catch (error) { console.error( "❌ El navegador bloqueó la reproducción:", error ); document.getElementById('currentSub').innerText = "⚠️ Pulsa reproducir nuevamente"; } } ); hlsInstance.on( Hls.Events.ERROR, function (event, data) { console.error( "❌ ERROR HLS:", data ); if (!data.fatal) { return; } switch (data.type) { case Hls.ErrorTypes.NETWORK_ERROR: console.warn( "⚠️ Error de red HLS. Reintentando..." ); try { hlsInstance.startLoad(); } catch (e) { console.error(e); } break; case Hls.ErrorTypes.MEDIA_ERROR: console.warn( "⚠️ Error de media HLS. Recuperando..." ); try { hlsInstance.recoverMediaError(); } catch (e) { console.error(e); } break; default: console.error( "❌ Error HLS fatal. Reiniciando reproductor." ); try { hlsInstance.destroy(); } catch (e) {} hlsInstance = null; break; } } ); return; } // ----------------------------------------- // HLS NATIVO // Safari / algunos dispositivos // ----------------------------------------- if ( audio.canPlayType( 'application/vnd.apple.mpegurl' ) ) { console.log( "📺 HLS nativo detectado" ); audio.src = url; audio.addEventListener( 'loadedmetadata', async function nativeHLSPlay() { audio.removeEventListener( 'loadedmetadata', nativeHLSPlay ); try { await audio.play(); console.log( "▶️ HLS nativo reproduciendo" ); document.getElementById('currentSub').innerText = `🔴 EN VIVO · ${s.city}, ${s.country}`; } catch (error) { console.error( "❌ Error HLS nativo:", error ); } } ); audio.load(); return; } // ----------------------------------------- // HLS NO COMPATIBLE // ----------------------------------------- console.error( "❌ Este navegador no soporta HLS." ); document.getElementById('currentSub').innerText = "⚠️ Este dispositivo no soporta este formato"; return; } // --------------------------------------------- // AUDIO DIRECTO MP3 / AAC / OGG // --------------------------------------------- console.log( "🎧 Configurando audio directo..." ); audio.src = url; audio.load(); try { await audio.play(); console.log( "▶️ AUDIO DIRECTO REPRODUCIENDO:", s.name ); document.getElementById('currentSub').innerText = `🔴 EN VIVO · ${s.city}, ${s.country}`; } catch (error) { console.error( "❌ Error reproduciendo audio directo:", error ); document.getElementById('currentSub').innerText = "⚠️ Pulsa reproducir nuevamente"; } }
